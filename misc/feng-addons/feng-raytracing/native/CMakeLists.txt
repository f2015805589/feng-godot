cmake_minimum_required(VERSION 3.30)
project(FengNRD LANGUAGES CXX)
include(FetchContent)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(GODOT_CPP_ROOT "" CACHE PATH "godot-cpp checkout with generated headers")
set(GODOT_CPP_LIBRARY "" CACHE FILEPATH "matching built godot-cpp static library")
if(NOT GODOT_CPP_ROOT OR NOT GODOT_CPP_LIBRARY)
    message(FATAL_ERROR "Set GODOT_CPP_ROOT and GODOT_CPP_LIBRARY to a built godot-cpp checkout")
endif()
set(NRD_STATIC_LIBRARY ON CACHE BOOL "" FORCE)
set(NRD_EMBEDS_DXIL_SHADERS OFF CACHE BOOL "" FORCE)
set(NRD_EMBEDS_DXBC_SHADERS OFF CACHE BOOL "" FORCE)
set(NRD_EMBEDS_SPIRV_SHADERS ON CACHE BOOL "" FORCE)
set(NRD_SUPPORTS_QUAD_INTRINSICS OFF CACHE BOOL "" FORCE)
set(NRD_NORMAL_ENCODING 2 CACHE STRING "" FORCE)
set(NRD_ROUGHNESS_ENCODING 1 CACHE STRING "" FORCE)
FetchContent_Declare(nrd
    URL https://github.com/NVIDIA-RTX/NRD/archive/e06852bd545973cb224eb28820b891205cfc1624.zip
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(nrd)
add_library(feng_nrd SHARED src/nrd_bridge.cpp)
target_compile_features(feng_nrd PRIVATE cxx_std_17)
target_compile_definitions(feng_nrd PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN TYPED_METHOD_BIND)
target_include_directories(feng_nrd PRIVATE "${GODOT_CPP_ROOT}/include" "${GODOT_CPP_ROOT}/gen/include" "${GODOT_CPP_ROOT}/gdextension")
target_link_libraries(feng_nrd PRIVATE NRD "${GODOT_CPP_LIBRARY}")
set_target_properties(feng_nrd PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/../bin")
