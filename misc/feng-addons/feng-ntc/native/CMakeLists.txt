cmake_minimum_required(VERSION 3.22)
project(feng-ntc LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This first LibNTC binary integration targets Windows x64")
endif()
set(FENG_GODOT_CPP "${CMAKE_CURRENT_SOURCE_DIR}/../../feng-idweight-terrain/native/godot-cpp" CACHE PATH "godot-cpp SDK (override for standalone builds)")
set(FENG_NTC_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/../.sdk/include" CACHE PATH "Pinned LibNTC headers")
set(FENG_CVTT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../thirdparty/cvtt" CACHE PATH "Convection Texture Tools source (build dependency only)")
if(NOT EXISTS "${FENG_NTC_HEADERS}/libntc/ntc.h")
    message(FATAL_ERROR "Run python tools/setup_sdk.py first")
endif()
set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
set(GODOTCPP_PRECISION single CACHE STRING "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "" FORCE)
set(GODOTCPP_USE_HOT_RELOAD OFF CACHE BOOL "" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "" FORCE)
add_subdirectory("${FENG_GODOT_CPP}" "${CMAKE_CURRENT_BINARY_DIR}/godot-cpp")
if(NOT EXISTS "${FENG_CVTT}/ConvectionKernels_SingleFile.cpp")
    message(FATAL_ERROR "Set FENG_CVTT to the Convection Texture Tools source directory")
endif()
add_library(feng-ntc-bc7 STATIC "${FENG_CVTT}/ConvectionKernels_SingleFile.cpp")
target_compile_definitions(feng-ntc-bc7 PRIVATE CVTT_SINGLE_FILE)
target_compile_options(feng-ntc-bc7 PRIVATE /w)
add_library(feng-ntc SHARED feng_ntc.cpp)
target_include_directories(feng-ntc PRIVATE "${FENG_NTC_HEADERS}" "${FENG_CVTT}")
target_link_libraries(feng-ntc PRIVATE godot-cpp feng-ntc-bc7)
target_compile_features(feng-ntc PRIVATE cxx_std_17)
target_compile_definitions(feng-ntc PRIVATE NTC_STATIC=1 WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(feng-ntc PRIVATE /utf-8 /W3)
set_target_properties(feng-ntc PROPERTIES PREFIX "" OUTPUT_NAME "feng_ntc"
    RUNTIME_OUTPUT_DIRECTORY "$<1:${CMAKE_CURRENT_SOURCE_DIR}/../bin>")
